package main

import (
	"fmt"
	"os"
	"path/filepath"
	"sync"
	"testing"
)

func commit(dir, target, content string) error {
	f, err := os.CreateTemp(dir, "stage-*.tmp")
	if err != nil {
		return err
	}
	fmt.Fprint(f, content)
	f.Sync()
	n := f.Name()
	f.Close()
	err = os.Link(n, filepath.Join(dir, target))
	os.Remove(n)
	return err
}

// agy's claim: two DIFFERENT terminal pathnames both succeed.
func TestSplitBrain(t *testing.T) {
	both := 0
	for i := 0; i < 200; i++ {
		dir := t.TempDir()
		var wg sync.WaitGroup
		start := make(chan struct{})
		wg.Add(2)
		go func() { defer wg.Done(); <-start; commit(dir, "result.txt", "the answer") }()
		go func() { defer wg.Done(); <-start; commit(dir, "publish.reject", "timed out") }()
		close(start)
		wg.Wait()
		_, e1 := os.Stat(filepath.Join(dir, "result.txt"))
		_, e2 := os.Stat(filepath.Join(dir, "publish.reject"))
		if e1 == nil && e2 == nil {
			both++
		}
	}
	fmt.Printf("SPLIT-BRAIN: both terminal records present in %d/200 runs\n", both)
}

// The fix: a single create-if-absent linearisation point.
func TestSingleOutcome(t *testing.T) {
	bad := 0
	for i := 0; i < 200; i++ {
		dir := t.TempDir()
		var wg sync.WaitGroup
		start := make(chan struct{})
		wg.Add(2)
		// each writer stages its payload, then races for the ONE outcome name
		go func() {
			defer wg.Done()
			<-start
			commit(dir, "result.txt", "the answer")
			commit(dir, "outcome.json", `{"verdict":"committed"}`)
		}()
		go func() {
			defer wg.Done()
			<-start
			commit(dir, "publish.reject", "timed out")
			commit(dir, "outcome.json", `{"verdict":"rejected"}`)
		}()
		close(start)
		wg.Wait()
		b, err := os.ReadFile(filepath.Join(dir, "outcome.json"))
		if err != nil || (string(b) != `{"verdict":"committed"}` && string(b) != `{"verdict":"rejected"}`) {
			bad++
		}
	}
	fmt.Printf("SINGLE OUTCOME: ambiguous/missing verdict in %d/200 runs\n", bad)
}
